```
**PROCESS**:
```
# Python implementation (if PDF)
import fitz  # PyMuPDF
import json
import hashlib

def extract_data(file_path: str) -> list[dict]:
    """
    Extract structured data from PDF or JSON
    Returns: [{id, title, content, metadata}, ...]
    """
    cache_key = hashlib.md5(open(file_path, 'rb').read()).hexdigest()
    cache_file = f".cache/{cache_key}.json"
    
    # Check cache first
    if os.path.exists(cache_file):
        with open(cache_file) as f:
            return json.load(f)
    
    if file_path.endswith('.pdf'):
        doc = fitz.open(file_path)
        data = []
        for page_num, page in enumerate(doc):
            data.append({
                "id": f"page_{page_num}",
                "title": extract_title(page),
                "content": page.get_text(),
                "metadata": {"page": page_num, "words": len(page.get_text().split())}
            })
        doc.close()
    elif file_path.endswith('.json'):
        with open(file_path) as f:
            data = json.load(f)
    else:
        raise ValueError("Unsupported file type")
    
    # Cache result
    os.makedirs(".cache", exist_ok=True)
    with open(cache_file, 'w') as f:
        json.dump(data, f)
    
    return data

# FastAPI example
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
import json

app = FastAPI()

class Item(BaseModel):
    id: str
    title: str
    content: str
    metadata: dict = {}

@app.get("/api/items")
async def list_items():
    with open("data/processed.json") as f:
        return json.load(f)

@app.get("/api/items/{item_id}")
async def get_item(item_id: str):
    with open("data/processed.json") as f:
        items = json.load(f)
    
    item = next((i for i in items if i['id'] == item_id), None)
    if not item:
        raise HTTPException(status_code=404, detail="Item not found")
    
    return item
