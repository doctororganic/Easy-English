## Feature 1: File Upload & Text Reading

### Simple Upload + Text Extraction
**Best Tools:**
- **PDF.js** (Mozilla) - Lightweight, no backend needed, client-side parsing[1][2]
- **Mammoth.js** - For Word docs (.docx) - 25KB gzipped, browser-based[3]
- **Papa Parse** - CSV/TSV files, fast & reliable, 25KB[4]

**Implementation (Basic):**
```javascript
// PDF text extraction (client-side)
import * as pdfjsLib from 'pdfjs-dist';

async function extractPDFText(file) {
  try {
    const arrayBuffer = await file.arrayBuffer();
    const pdf = await pdfjsLib.getDocument(arrayBuffer).promise;
    let fullText = '';
    
    for (let i = 1; i <= pdf.numPages; i++) {
      const page = await pdf.getPage(i);
      const content = await page.getTextContent();
      fullText += content.items.map(item => item.str).join(' ');
    }
    return fullText;
  } catch (error) {
    console.error('PDF parse failed:', error);
    return 'Error reading file. Ensure it\'s a valid PDF.';
  }
}
```


**Error Prevention Tips:**
- **File size limit:** Check `file.size < 10MB` before processing[4]
- **File type validation:** `['application/pdf', 'text/plain'].includes(file.type)`[4]
- **Timeout handling:** Set 30s timeout for large files[4]

## Feature 2: Grammar/Vocab Canvas Visualizations

### Lightweight Canvas Libraries
**Recommended Tools:**
- **Fabric.js** (60KB) - Interactive canvas, drag-drop objects, export to image[5]
- **Konva.js** (80KB) - Simple API, great for educational diagrams, mobile-friendly[5]
- **Rough.js** (35KB) - Hand-drawn style charts (attractive for students)[5]

**Quick Implementation:**
```javascript
// Grammar diagram with Konva.js
import Konva from 'konva';

function createGrammarCard(word, definition) {
  const stage = new Konva.Stage({
    container: 'canvas-container',
    width: 400,
    height: 200
  });
  
  const layer = new Konva.Layer();
  
  // Background
  layer.add(new Konva.Rect({
    width: 400, height: 200,
    fill: '#f0f8ff', cornerRadius: 10
  }));
  
  // Word (large text)
  layer.add(new Konva.Text({
    text: word, x: 20, y: 30,
    fontSize: 32, fontStyle: 'bold', fill: '#2c3e50'
  }));
  
  // Definition (smaller text)
  layer.add(new Konva.Text({
    text: definition, x: 20, y: 80,
    fontSize: 18, width: 360, fill: '#34495e'
  }));
  
  stage.add(layer);
  return stage;
}
```


**Error-Free Tips:**
- **Memory cleanup:** Call `stage.destroy()` when switching cards[5]
- **Mobile scaling:** Use `width: window.innerWidth * 0.9` for responsiveness[5]
- **Font loading:** Wait for fonts with `document.fonts.ready.then(...)` before rendering[6]

## Feature 3: Convert to Portable Image

### Best Conversion Libraries (Comparison)
| Library | Size | Speed | Quality | Best For |
|---------|------|-------|---------|----------|
| **html-to-image** | 15KB | Fast | Good | Simple cards, small RAM [6][7] |
| **dom-to-image-more** | 30KB | Fast | Excellent | Complex layouts, best quality [6][7] |
| html2canvas | 110KB | Slower | Variable | Legacy support only [6][7] |

**Recommended: html-to-image** (lightest + reliable)[7][6]

**Implementation:**
```javascript
import { toPng, toJpeg } from 'html-to-image';

async function exportCardAsImage(elementId) {
  const element = document.getElementById(elementId);
  
  try {
    // Add loading state
    element.classList.add('exporting');
    
    // Convert to image
    const dataUrl = await toPng(element, {
      quality: 0.95,
      pixelRatio: 2 // Retina quality
    });
    
    // Download or share
    const link = document.createElement('a');
    link.download = `vocab-card-${Date.now()}.png`;
    link.href = dataUrl;
    link.click();
    
    element.classList.remove('exporting');
    return dataUrl;
  } catch (error) {
    console.error('Export failed:', error);
    alert('Image export failed. Try again or reduce card complexity.');
    element.classList.remove('exporting');
  }
}
```


**Error Prevention (Critical):**
- **CORS issues:** Host fonts/images on same domain or use `crossOrigin="anonymous"`[10][7]
- **CSS styles:** Inline critical styles; external CSS may not render[7]
- **Canvas size limits:** Keep cards under 4096×4096px (browser limit)[6]
- **Memory leaks:** Call `URL.revokeObjectURL(dataUrl)` after use[6]

## Error Management Summary

**Prevent Common Issues:**
1. **Wrap all async operations in try-catch**[1][3]
2. **Show loading states** during file processing[2]
3. **Validate inputs** before processing (file type, size)[3][4]
4. **Set timeouts** for long operations (30s max)[4]
5. **Clear memory** after conversions (`stage.destroy()`, `URL.revokeObjectURL()`)[6][5]
6. **Test on mobile** - canvas/image export uses more RAM there[7][6]

Web Speech API - Native browser TTS, no library needed​

javascript
// Simple TTS function - works immediately
function speakText(text, options = {}) {
  // Check browser support
  if (!('speechSynthesis' in window)) {
    alert('Your browser doesn\'t support text-to-speech.');
    return;
  }
  
  const utterance = new SpeechSynthesisUtterance(text);
  
  // Configure voice settings
  utterance.lang = options.lang || 'en-US'; // en-GB, en-AU, etc.
  utterance.rate = options.rate || 0.9; // Speed (0.1-10, default 1)
  utterance.pitch = options.pitch || 1; // Pitch (0-2, default 1)
  utterance.volume = options.volume || 1; // Volume (0-1)
  
  // Error handling
  utterance.onerror = (event) => {
    console.error('Speech error:', event.error);
    if (event.error === 'interrupted') {
      // User stopped it, normal behavior
      return;
    }
    alert('Voice playback failed. Try refreshing the page.');
  };
  
  // Success callback
  utterance.onend = () => {
    console.log('Speech finished');
  };
  
  // Speak!
  speechSynthesis.speak(utterance);
}

// Usage examples
speakText('[translate:Hello, how are you today?]');
speakText('[translate:The cat sat on the mat.]', { 
  rate: 0.7, // Slower for learners
  lang: 'en-US' 
});

function getAvailableVoices() {
  return new Promise((resolve) => {
    let voices = speechSynthesis.getVoices();
    
    if (voices.length) {
      resolve(voices);
    } else {
      // Some browsers load voices asynchronously
      speechSynthesis.onvoiceschanged = () => {
        voices = speechSynthesis.getVoices();
        resolve(voices);
      };
    }
  });
}

// Use it
async function setupVoiceSelector() {
  const voices = await getAvailableVoices();
  const englishVoices = voices.filter(v => v.lang.startsWith('en'));
  
  console.log('Available English voices:');
  englishVoices.forEach((voice, i) => {
    console.log(`${i}: ${voice.name} (${voice.lang})`);
  });
  
  return englishVoices;
}

// Select specific voice
async function speakWithVoice(text, voiceName = 'Google US English') {
  const voices = await getAvailableVoices();
  const utterance = new SpeechSynthesisUtterance(text);
  
  const selectedVoice = voices.find(v => v.name.includes(voiceName));
  if (selectedVoice) {
    utterance.voice = selectedVoice;
  }
  
  speechSynthesis.speak(utterance);
}

Error-Free Implementation Tips
1. Always Check Browser Support
javascript
function checkVoiceSupport() {
  const features = {
    tts: 'speechSynthesis' in window,
    stt: 'SpeechRecognition' in window || 'webkitSpeechRecognition' in window,
    webgpu: 'gpu' in navigator,
    wasm: typeof WebAssembly === 'object'
  };
  
  console.log('Voice features available:', features);
  
  if (!features.tts && !features.wasm) {
    alert('Your browser doesn\'t support voice features. Use Chrome, Edge, or Safari.');
  }
  
  return features;
}

4. Manage Long Text
javascript
// Split long text into chunks (browsers have limits)
function speakLongText(text, chunkSize = 200) {
  const chunks = text.match(new RegExp(`.{1,${chunkSize}}`, 'g'));
  let index = 0;
  
  function speakNextChunk() {
    if (index >= chunks.length) return;
    
    const utterance = new SpeechSynthesisUtterance(chunks[index]);
    utterance.onend = () => {
      index++;
      speakNextChunk();
    };
    
    speechSynthesis.speak(utterance);
  }
  
  speakNextChunk();
}
