from fastapi.responses import JSONResponse

app = FastAPI(title="PDF Extraction API")

# ==================== MODELS ====================

class DocumentMetadata(BaseModel):
    title: Optional[str]
    author: Optional[str]
    total_pages: int
    file_size: int

class PageContent(BaseModel):
    page_number: int
    text: str
    word_count: int

class TableData(BaseModel):
    page_number: int
    headers: List[str]
    rows: List[List[str]]
    structured_data: List[dict]

class AnnotationData(BaseModel):
    page_number: int
    type: str
    author: str
    contents: str
    highlighted_text: Optional[str]

class DocumentResponse(BaseModel):
    document_id: str
    metadata: DocumentMetadata
    statistics: dict
    extracted_at: str

# ==================== BATCH PROCESSING ====================

@app.post("/api/v1/documents/batch")
async def upload_multiple_pdfs(
    files: List[UploadFile] = File(...),
    background_tasks: BackgroundTasks = None
):
    """
    Upload multiple PDFs for batch processing
    """
    job_id = str(uuid.uuid4())
    file_paths = []
    
    for file in files:
        if file.filename.endswith('.pdf'):
            file_path = f"/tmp/{job_id}_{file.filename}"
            with open(file_path, "wb") as f:
                content = await file.read()
                f.write(content)
            file_paths.append(file_path)
    
    # Add to processing queue
    add_to_queue(job_id, file_paths)
    
    return {
        "job_id": job_id,
        "files_queued": len(file_paths),
        "status": "processing"
    }

@app.get("/api/v1/jobs/{job_id}/status")
